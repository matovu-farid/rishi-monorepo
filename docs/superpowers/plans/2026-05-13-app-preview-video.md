# Mac App Store Preview Video Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a `pnpm run preview:record` workflow in `apps/rishi-electron` that produces a 25s, Apple-compliant Mac App Store preview video (and an optional OpenAI-narrated variant) by driving the Electron app with Playwright.

**Architecture:** A new `marketing/` directory next to `e2e/`. An orchestrator script launches the app via the existing `e2e/helpers/electron-app.ts`, installs Playwright mocks for the chat endpoint, seeds the library with fixture books, and runs four "beat" modules in sequence. Playwright's built-in video recorder writes `.webm` files; a postprocess step uses system `ffmpeg` to concat (if multiple windows were captured), transcode to H.264, and optionally mux narrated audio generated by OpenAI TTS. An `ffprobe` audit verifies the output meets Apple's specs before exit.

**Tech Stack:** TypeScript (tsx for direct execution), Playwright (`_electron`), Electron, system `ffmpeg` + `ffprobe`, OpenAI SDK (`gpt-4o-mini-tts`), vitest for unit tests where useful.

**Spec:** `docs/superpowers/specs/2026-05-13-app-preview-video-design.md`

// apps/rishi-electron/marketing/preflight.ts
import { execFileSync } from 'child_process'
import fs from 'fs'
import path from 'path'

export function checkBinaryOnPath(name: string): boolean {
  try {
    execFileSync('which', [name], { stdio: 'ignore' })
    return true
  } catch {
    return false
  }
}

export function checkBuildArtifacts(outDir: string): boolean {
  return fs.existsSync(path.join(outDir, 'main', 'index.js'))
}

export function preflight(): void {
  if (!checkBinaryOnPath('ffmpeg')) {
    console.error('Error: ffmpeg is not installed or not on PATH.')
    console.error('Install with: brew install ffmpeg')
    process.exit(1)
  }
  if (!checkBinaryOnPath('ffprobe')) {
    console.error('Error: ffprobe is not on PATH (usually ships with ffmpeg).')
    process.exit(1)
  }
  const outDir = path.resolve(__dirname, '..', 'out')
  if (!checkBuildArtifacts(outDir)) {
    console.error(`Error: built app not found at ${outDir}/main/index.js`)
    console.error('Run: pnpm run build')
    process.exit(1)
  }
}
```

// apps/rishi-electron/marketing/cli.ts
export interface CliArgs {
  narrate: boolean
  beats: number[] | null
  keepWebm: boolean
  voice: string
  openWhenDone: boolean
}

export function parseArgs(argv: string[]): CliArgs {
  const args: CliArgs = {
    narrate: false,
    beats: null,
    keepWebm: false,
    voice: 'nova',
    openWhenDone: true
  }
  for (const a of argv) {
    if (a === '--narrate') args.narrate = true
    else if (a === '--keep-webm') args.keepWebm = true
    else if (a === '--no-open') args.openWhenDone = false
    else if (a.startsWith('--beats=')) {
      args.beats = a.slice('--beats='.length).split(',').map((s) => parseInt(s, 10))
    } else if (a.startsWith('--voice=')) {
      args.voice = a.slice('--voice='.length)
    } else {
      throw new Error(`unknown argument: ${a}`)
    }
  }
  return args
}
```

// apps/rishi-electron/marketing/mocks/chat-response.ts
import type { Page, Route } from '@playwright/test'

/** Pre-canned answer that streams in Beat 4. */
export const MOCK_ANSWER =
  'This chapter explores transformation through adversity — the protagonist\'s choices echo earlier scenes in chapters 3 and 7. Notice how the imagery shifts from cold to warm.'

/**
 * URL pattern matched by Playwright's route handler. MUST match the result
 * recorded in marketing/README.md under "Chat endpoint" from Task 6.
 */
const CHAT_URL_GLOB = '**/api/chat/**'

export async function installChatMock(page: Page): Promise<{ wasHit: () => boolean }> {
  let hit = false

  await page.route(CHAT_URL_GLOB, async (route: Route) => {
    hit = true
    const words = MOCK_ANSWER.split(' ')
    const chunkDelayMs = Math.floor(4000 / words.length)

    // Build a single SSE response body, with delays between chunks so the
    // body is "ready" only after ~4 seconds — matching real stream pacing.
    const encoder = (text: string): string => `data: ${JSON.stringify({ delta: text })}\n\n`
    const body = await new Promise<string>((resolve) => {
      let out = ''
      let i = 0
      const tick = (): void => {
        if (i >= words.length) {
          out += 'data: [DONE]\n\n'
          resolve(out)
          return
        }
        out += encoder(words[i] + (i < words.length - 1 ? ' ' : ''))
        i++
        setTimeout(tick, chunkDelayMs)
      }
      tick()
    })

    await route.fulfill({
      status: 200,
      contentType: 'text/event-stream',
      body
    })
  })

  return { wasHit: () => hit }
}
```

// apps/rishi-electron/marketing/ffprobe-audit.ts
import { execFileSync } from 'child_process'
import { APPLE_LIMITS } from './config'

export interface Probed {
  codec: string
  width: number
  height: number
  fps: number
  durationSec: number
  sizeBytes: number
  containerHasMov: boolean
}

export function parseFfprobeJson(json: string): Probed {
  const data = JSON.parse(json)
  const video = data.streams.find((s: { codec_type: string }) => s.codec_type === 'video')
  if (!video) throw new Error('ffprobe: no video stream found')
  const rate = video.avg_frame_rate ?? video.r_frame_rate ?? '0/1'
  const [num, den] = String(rate)
    .split('/')
    .map((n: string) => parseInt(n, 10))
  return {
    codec: video.codec_name,
    width: video.width,
    height: video.height,
    fps: den ? num / den : 0,
    durationSec: parseFloat(data.format.duration),
    sizeBytes: parseInt(data.format.size, 10),
    containerHasMov: String(data.format.format_name ?? '').includes('mov')
  }
}

export function validate(p: Probed): void {
  if (p.codec !== APPLE_LIMITS.codec) {
    throw new Error(`Wrong video codec: got '${p.codec}', want '${APPLE_LIMITS.codec}'`)
  }
  if (p.width !== 1920 || p.height !== 1080) {
    throw new Error(`Wrong resolution: got ${p.width}x${p.height}, want 1920x1080`)
  }
  if (Math.abs(p.fps - APPLE_LIMITS.fps) > 0.01) {
    throw new Error(`Wrong fps: got ${p.fps}, want ${APPLE_LIMITS.fps}`)
  }
  if (p.durationSec < APPLE_LIMITS.minDurationSec || p.durationSec > APPLE_LIMITS.maxDurationSec) {
    throw new Error(
      `Duration ${p.durationSec}s outside Apple's [${APPLE_LIMITS.minDurationSec}, ${APPLE_LIMITS.maxDurationSec}]s window`
    )
  }
  if (p.sizeBytes > APPLE_LIMITS.maxBytes) {
    throw new Error(`File size ${p.sizeBytes} exceeds Apple's max ${APPLE_LIMITS.maxBytes}`)
  }
  if (!p.containerHasMov) {
    throw new Error(`Wrong container: format_name does not include 'mov'`)
  }
}

export function audit(filePath: string): void {
  const json = execFileSync('ffprobe', [
    '-v', 'quiet',
    '-print_format', 'json',
    '-show_format',
    '-show_streams',
    filePath
  ]).toString()
  validate(parseFfprobeJson(json))
}

// apps/rishi-electron/marketing/postprocess.ts
import { execFileSync } from 'child_process'
import fs from 'fs'
import path from 'path'
import { CRF, FPS, RESOLUTION } from './config'

export function buildConcatArgs(listFile: string, out: string): string[] {
  return ['-y', '-f', 'concat', '-safe', '0', '-i', listFile, '-c', 'copy', out]
}

export function buildSilentTranscodeArgs(input: string, output: string): string[] {
  return [
    '-y', '-i', input,
    '-r', String(FPS),
    '-vf', `scale=${RESOLUTION.width}:${RESOLUTION.height}:flags=lanczos`,
    '-c:v', 'libx264',
    '-pix_fmt', 'yuv420p',
    '-profile:v', 'high',
    '-crf', String(CRF),
    '-movflags', '+faststart',
    '-an',
    output
  ]
}

export interface MuxInput {
  videoIn: string
  audioFiles: string[]
  offsetsMs: number[]
  out: string
}

export function buildNarratedMuxArgs(m: MuxInput): string[] {
  const inputs: string[] = ['-y', '-i', m.videoIn]
  for (const f of m.audioFiles) inputs.push('-i', f)

  const delayLabels: string[] = []
  const filters: string[] = []
  m.audioFiles.forEach((_, i) => {
    const off = m.offsetsMs[i]
    const label = `[a${i + 1}]`
    filters.push(`[${i + 1}:a]adelay=${off}|${off}${label}`)
    delayLabels.push(label)
  })
  filters.push(`${delayLabels.join('')}amix=inputs=${m.audioFiles.length}:duration=longest[aout]`)

  return [
    ...inputs,
    '-filter_complex', filters.join('; '),
    '-map', '0:v', '-map', '[aout]',
    '-c:v', 'copy',
    '-c:a', 'aac', '-b:a', '192k',
    m.out
  ]
}

export function writeConcatList(parts: string[], runDir: string): string {
  const listPath = path.join(runDir, 'parts.txt')
  const body = parts.map((p) => `file '${p.replace(/'/g, "'\\''")}'`).join('\n') + '\n'
  fs.writeFileSync(listPath, body)
  return listPath
}

export function ffmpeg(args: string[]): void {
  execFileSync('ffmpeg', args, { stdio: 'inherit' })
}

export interface PostprocessInput {
  webmParts: string[]
  runDir: string
  audioFiles?: string[]
  audioOffsetsMs?: number[]
  finalOut: string
}

export function postprocess(p: PostprocessInput): void {
  // 1) Concat (or single-copy if one part).
  let concatPath = p.webmParts[0]
  if (p.webmParts.length > 1) {
    const listFile = writeConcatList(p.webmParts, p.runDir)
    concatPath = path.join(p.runDir, 'concat.webm')
    ffmpeg(buildConcatArgs(listFile, concatPath))
  }

  // 2) Transcode to silent H.264 mov.
  const silentPath = p.audioFiles ? path.join(p.runDir, 'silent.mov') : p.finalOut
  ffmpeg(buildSilentTranscodeArgs(concatPath, silentPath))

  // 3) Optional audio mux.
  if (p.audioFiles && p.audioOffsetsMs) {
    ffmpeg(
      buildNarratedMuxArgs({
        videoIn: silentPath,
        audioFiles: p.audioFiles,
        offsetsMs: p.audioOffsetsMs,
        out: p.finalOut
      })
    )
  }
}

// apps/rishi-electron/marketing/narrate.ts
import { execFileSync } from 'child_process'
import crypto from 'crypto'
import fs from 'fs'
import path from 'path'
import OpenAI from 'openai'

export function cacheKey(text: string, voice: string, style: string): string {
  return crypto.createHash('sha256').update(`${voice}::${style}::${text}`).digest('hex').slice(0, 16)
}

export function audioPath(audioDir: string, key: string): string {
  return path.join(audioDir, `${key}.mp3`)
}

export interface NarrateInput {
  text: string
  voice: string
  style: string
  audioDir: string
}

/** Returns the path to a cached or freshly generated MP3. Skips the API call on cache hit. */
export async function narrate(input: NarrateInput): Promise<string> {
  const key = cacheKey(input.text, input.voice, input.style)
  const out = audioPath(input.audioDir, key)
  if (fs.existsSync(out)) return out

  if (!process.env.OPENAI_API_KEY) {
    throw new Error('OPENAI_API_KEY is required for --narrate mode (no cache hit)')
  }
  fs.mkdirSync(input.audioDir, { recursive: true })

  const client = new OpenAI()
  const response = await client.audio.speech.create({
    model: 'gpt-4o-mini-tts',
    voice: input.voice,
    input: input.text,
    instructions: input.style
  })
  const buf = Buffer.from(await response.arrayBuffer())
  fs.writeFileSync(out, buf)
  return out
}

export function getAudioDurationMs(filePath: string): number {
  const out = execFileSync('ffprobe', [
    '-v', 'quiet',
    '-print_format', 'json',
    '-show_format',
    filePath
  ]).toString()
  return Math.round(parseFloat(JSON.parse(out).format.duration) * 1000)
}

  preflight()

  const runDir = path.join(OUT_ROOT, `run-${Date.now()}`)
  fs.mkdirSync(runDir, { recursive: true })
  console.log(`[record] run dir: ${runDir}`)

  const launched = await launchApp({
    recordVideo: { dir: runDir, size: RESOLUTION }
  })

  // Resize the BrowserWindow to match the recordVideo size.
  await launched.app.evaluate(({ BrowserWindow }, size) => {
    const win = BrowserWindow.getFocusedWindow()
    if (!win) throw new Error('No focused window after launch')
    win.setSize(size.width, size.height)
    win.center()
  }, RESOLUTION)
  await launched.page.waitForTimeout(500)

  const chatMock = await installChatMock(launched.page)

  const seeded = await seedLibrary(launched.page)
  console.log(`[record] seeded library:`, seeded)

  const beatTimings: { id: string; ms: number }[] = []

  const t0 = Date.now()
  await runBeat01(launched.page)
  beatTimings.push({ id: '01-library', ms: Date.now() - t0 })

  const t1 = Date.now()
  const epubPage = await runBeat02(launched.page, seeded.epubId)
  beatTimings.push({ id: '02-epub-reader', ms: Date.now() - t1 })

  const t2 = Date.now()
  const pdfPage = await runBeat03(launched.page, epubPage, seeded.pdfId)
  beatTimings.push({ id: '03-pdf-reader', ms: Date.now() - t2 })

  const t3 = Date.now()
  await runBeat04(pdfPage)
  beatTimings.push({ id: '04-ai-chat', ms: Date.now() - t3 })

  if (!chatMock.wasHit()) {
    throw new Error('Beat 4 finished but the mocked chat route was never hit. Check Task 6 URL.')
  }

  console.log(`[record] beat timings:`, beatTimings)

  await closeApp(launched)
  // Let the filesystem settle so Playwright finishes flushing .webm files.
  await new Promise((r) => setTimeout(r, 500))

  const webms = fs
    .readdirSync(runDir)
    .filter((f) => f.endsWith('.webm'))
    .map((f) => path.join(runDir, f))
    .sort((a, b) => fs.statSync(a).mtimeMs - fs.statSync(b).mtimeMs)

  if (webms.length === 0) {
    throw new Error(`No .webm files produced in ${runDir}`)
  }
  console.log(`[record] captured ${webms.length} webm part(s)`)

  const finalOut = path.join(OUT_ROOT, 'rishi-app-preview.mov')
  postprocess({
    webmParts: webms,
    runDir,
    finalOut
  })
  console.log(`[record] wrote ${finalOut}`)

  audit(finalOut)
  console.log(`[record] ffprobe audit: PASS`)

  if (!args.keepWebm) {
    for (const w of webms) fs.rmSync(w, { force: true })
    fs.rmSync(path.join(runDir, 'concat.webm'), { force: true })
    fs.rmSync(path.join(runDir, 'parts.txt'), { force: true })
  }

  if (args.openWhenDone) {
    try {
      execFileSync('open', [finalOut])
    } catch {
      // Non-mac or `open` missing — non-fatal.
    }
  }
}

main().catch((err) => {
  console.error('[record] FAILED:', err)
  process.exit(1)
})
```
